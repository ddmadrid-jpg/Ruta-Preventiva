function renderTabla2(data) { const t2Stats = {}; data.forEach(d => { const cat = d['Tipo de Activo'] || d['Activos'] || d['Categoria'] || 'Post Mix'; const usr = d['Usuario'] || 'Sin Usuario'; const tec = d['Técnico'] || d['Tecnico'] || d['Jefe de Ventas'] || 'Sin Técnico'; const realizada = (d['Realizada 2'] || d['Realizada'] || d['Cumplimiento'] || '').toUpperCase(); if(!t2Stats[cat]) t2Stats[cat] = {}; if(!t2Stats[cat][usr]) t2Stats[cat][usr] = { tecnico: tec, SI: 0, NO: 0 }; if(realizada === "SI" || realizada === "1" || realizada === "EJECUTADA") t2Stats[cat][usr].SI++; else t2Stats[cat][usr].NO++; }); let html2 = ` `; let grandNO = 0, grandSI = 0; Object.keys(t2Stats).forEach(cat => { let catSI = 0, catNO = 0; // Ordenar usuarios de menor a mayor (STCYG01, STCYG02, etc.) const usuariosOrdenados = Object.keys(t2Stats[cat]).sort((a, b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' })); usuariosOrdenados.forEach(usr => { const uData = t2Stats[cat][usr]; const tot = uData.SI + uData.NO; catSI += uData.SI; catNO += uData.NO; const pctSI_num = tot > 0 ? (uData.SI / tot) * 100 : 0; const pctNO_num = tot > 0 ? (uData.NO / tot) * 100 : 0; const pctNO = pctNO_num.toFixed(1) + '%'; const pctSI = pctSI_num.toFixed(1) + '%'; // Alerta roja si el cumplimiento SI es menor al 92% const alertClass = pctSI_num < 92 ? 'alert-red' : ''; html2 += ` `; }); grandSI += catSI; grandNO += catNO; const catTot = catSI + catNO; const catPctNO = catTot > 0 ? ((catNO / catTot) * 100).toFixed(1) + '%' : '0.0%'; const catPctSI = catTot > 0 ? ((catSI / catTot) * 100).toFixed(1) + '%' : '0.0%'; html2 += ` `; }); const grandALL = grandSI + grandNO; const grandPctNO = grandALL > 0 ? ((grandNO / grandALL) * 100).toFixed(1) + '%' : '0.0%'; const grandPctSI = grandALL > 0 ? ((grandSI / grandALL) * 100).toFixed(1) + '%' : '0.0%'; html2 += `
Categoría Usuario Técnico NO (Cant. / %) SI (Cant. / %) Total Servicios
• ${cat} • ${usr} ${uData.tecnico} ${uData.NO} (${pctNO}) ${uData.SI} (${pctSI}) ${tot} (100%)
Total ${cat} ${catNO} (${catPctNO}) ${catSI} (${catPctSI}) ${catTot} (100%)
Total General ${grandNO} (${grandPctNO}) ${grandSI} (${grandPctSI}) ${grandALL} (100%)
`; document.getElementById('tabla-2-container').innerHTML = html2; }
